Section 03

Database Structure

Relational schema with polymorphic relationships and comprehensive audit trails.

Key Tables

-- Core Users
CREATE TABLE users_member (id INT PRIMARY KEY AUTO_INCREMENT, email VARCHAR(255) UNIQUE, phone VARCHAR(20), password_hash VARCHAR(255), is_email_verified BOOLEAN);

-- NFC Access Control (Polymorphic)
CREATE TABLE access_control_nfcdevice (id INT PRIMARY KEY, uid VARCHAR(16) UNIQUE, device_type ENUM('card','phone'), role ENUM('member','coach','admin'), member_id INT, coach_id INT, is_active BOOLEAN);

-- Access Logs
CREATE TABLE access_control_accesslog (id INT PRIMARY KEY, nfc_device_id INT, timestamp TIMESTAMP, allowed BOOLEAN, reason VARCHAR(255));

-- Course Booking
CREATE TABLE courses_coursebooking (id INT PRIMARY KEY, course_id INT, member_id INT, status ENUM('pending','confirmed'), payment_status ENUM('pending','paid'));
