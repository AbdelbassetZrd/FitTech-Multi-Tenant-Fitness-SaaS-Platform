3. Database Structure

Key Tables

-- Core Users
CREATE TABLE users_member (id INT PRIMARY KEY AUTO_INCREMENT, email VARCHAR(255) UNIQUE, phone VARCHAR(20), password_hash VARCHAR(255), is_email_verified BOOLEAN, created_at TIMESTAMP);

-- Subscriptions with Polymorphic Plans
CREATE TABLE subscriptions_plan (id INT PRIMARY KEY, name VARCHAR(100), plan_type ENUM('monthly','annual','per_session'), price DECIMAL(10,2), duration_days INT);

CREATE TABLE subscriptions_subscription (id INT PRIMARY KEY, user_id INT, plan_id INT, status ENUM('active','expired','cancelled'), sessions_remaining INT, FOREIGN KEY (user_id) REFERENCES users_member(id));

-- NFC Access Control (Polymorphic)
CREATE TABLE access_control_nfcdevice (id INT PRIMARY KEY, uid VARCHAR(16) UNIQUE, device_type ENUM('card','phone'), role ENUM('member','coach','admin'), member_id INT NULL, coach_id INT NULL, admin_user_id INT NULL, is_active BOOLEAN, FOREIGN KEY (member_id) REFERENCES users_member(id), CHECK ((member_id IS NOT NULL AND coach_id IS NULL) OR ...));

-- Access Logs
CREATE TABLE access_control_accesslog (id INT PRIMARY KEY, nfc_device_id INT, timestamp TIMESTAMP, allowed BOOLEAN, reason VARCHAR(255), FOREIGN KEY (nfc_device_id) REFERENCES access_control_nfcdevice(id));

-- Course Booking
CREATE TABLE courses_coursebooking (id INT PRIMARY KEY, course_id INT, member_id INT, status ENUM('pending','confirmed','cancelled'), payment_status ENUM('pending','paid'), UNIQUE(course_id, member_id, DATE(booked_at)));

-- AI Workout Logs
CREATE TABLE workouts_workoutlog (id INT PRIMARY KEY, member_id INT, exercise_name VARCHAR(100), pose_accuracy_score DECIMAL(3,2), detected_landmarks JSON, logged_at TIMESTAMP);

-- Chargily Payments
CREATE TABLE payments_transaction (id INT PRIMARY KEY, chargily_checkout_id VARCHAR(255), amount DECIMAL(10,2), currency VARCHAR(3), status ENUM('pending','paid','failed'), created_at TIMESTAMP);

5. Docker Implementation

Multi-Container docker-compose.yml

version: '3.8'
services:
  mysql:
    image: mysql:8.0
    container_name: fittech_mysql
    environment:
      MYSQL_ROOT_PASSWORD: ${DB_ROOT_PASSWORD}
      MYSQL_DATABASE: fittech_db
      MYSQL_USER: fittech_user
      MYSQL_PASSWORD: ${DB_PASSWORD}
    volumes:
      - mysql_data:/var/lib/mysql
    ports:
      - "3308:3306"
    networks:
      - fittech_network

  backend:
    build: ./backend
    container_name: fittech_backend
    environment:
      DEBUG: "False"
      DB_HOST: mysql
      JWT_SECRET_KEY: ${JWT_SECRET}
    depends_on:
      - mysql
    ports:
      - "8000:8000"

  frontend:
    build: ./frontend
    container_name: fittech_frontend
    ports:
      - "3000:3000"
    environment:
      NEXT_PUBLIC_API_URL: http://backend:8000

volumes:
  mysql_data:

networks:
  fittech_network:
    driver: bridge

Docker Benefits

  • Isolation: Each service in its own container
  • Portability: Same environment everywhere
  • Scalability: Easy horizontal scaling
  • Consistency: No "works on my machine" issues
